function CardEditor({
  card,
  onSave,
  onClose,
}: {
  card?: LibraryCard;
  onSave: (card: LibraryCard) => void;
  onClose: () => void;
}) {
  const [question, setQuestion] = useState(card?.question ?? "");
  const [answer, setAnswer] = useState(card?.answer ?? "");
  const [type, setType] = useState<CardKind>(card?.type ?? "qa");
  const [tags, setTags] = useState(card?.tags.join(", ") ?? "");

  const save = () => {
    if (!question.trim() || !answer.trim()) return;
    onSave({
      id: card?.id ?? `card.${Date.now()}`,
      version: card ? card.version + 1 : 1,
      question: question.trim(),
      answer: answer.trim(),
      type,
      tags: tags
        .split(",")
        .map((tag) => tag.trim())
        .filter(Boolean),
    });
  };

  return (
    <Modal title={card ? "Edit Card" : "Create Card"} onClose={onClose}>
      <div className="modal-body editor-form">
        <label className="field">
          <span>Type</span>
          <span className="select-wrap">
            <select value={type} onChange={(event) => setType(event.target.value as CardKind)}>
              <option value="qa">qa</option>
              <option value="formula">formula</option>
              <option value="cloze">cloze</option>
              <option value="multiple_choice">multiple_choice</option>
              <option value="true_false">true_false</option>
              <option value="numeric">numeric</option>
            </select>
            <ChevronDown size={18} />
          </span>
        </label>
        <label className="field">
          <span>Question</span>
          <textarea value={question} onChange={(event) => setQuestion(event.target.value)} />
        </label>
        <label className="field">
          <span>Answer</span>
          <textarea value={answer} onChange={(event) => setAnswer(event.target.value)} />
        </label>
        <label className="field">
          <span>Tags</span>
          <input
            value={tags}
            onChange={(event) => setTags(event.target.value)}
            placeholder="Machine Learning, Decision Tree"
          />
          <small>Separate Tags with commas.</small>
        </label>
      </div>
      <div className="modal-actions">
        <button className="button ghost" onClick={onClose}>
          Cancel
        </button>
        <button
          className="button primary"
          onClick={save}
          disabled={!question.trim() || !answer.trim()}
        >
          {card ? "Save changes" : "Create Card"}
        </button>
      </div>
    </Modal>
  );
}

function VirtualCardList({
  items,
  onSelect,
}: {
  items: LibraryCard[];
  onSelect: (card: LibraryCard) => void;
}) {
  const rowHeight = 104;
  const viewportHeight = 468;
  const overscan = 3;
  const [scrollTop, setScrollTop] = useState(0);
  const start = Math.max(0, Math.floor(scrollTop / rowHeight) - overscan);
  const visibleCount = Math.ceil(viewportHeight / rowHeight) + overscan * 2;
  const visible = items.slice(start, start + visibleCount);

  return (
    <div
      className="virtual-list"
      onScroll={(event) => setScrollTop(event.currentTarget.scrollTop)}
      aria-label="Card library"
    >
      <div className="virtual-spacer" style={{ height: items.length * rowHeight }}>
        {visible.map((card, offset) => (
          <button
            className="library-card-row"
            key={card.id}
            onClick={() => onSelect(card)}
            style={{ transform: `translateY(${(start + offset) * rowHeight}px)` }}
          >
            <div className="library-card-copy">
              <strong>{card.question}</strong>
              <span className="card-id">{card.id}</span>
              <div className="row-tags">
                <span>{card.type}</span>
                {card.tags.slice(0, 2).map((tag) => (
                  <span key={tag}>{tag}</span>
                ))}
              </div>
            </div>
            <ArrowRight size={18} />
          </button>
        ))}
      </div>
    </div>
  );
}

function CardsScreen({
  language,
  cards,
  setCards,
  initialIntent,
}: {
  language: Language;
  cards: LibraryCard[];
  setCards: React.Dispatch<React.SetStateAction<LibraryCard[]>>;
  initialIntent: CardsIntent;
}) {
  const t = copy[language];
  const [search, setSearch] = useState("");
  const [type, setType] = useState("all");
  const [tag, setTag] = useState("all");
  const [selected, setSelected] = useState<LibraryCard | null>(null);
  const [editorCard, setEditorCard] = useState<LibraryCard | "new" | null>(
    initialIntent === "create" ? "new" : null,
  );
  const [confirmClear, setConfirmClear] = useState(false);
  const [uploadResult, setUploadResult] = useState<"none" | "success" | "error">("none");
  const [notice, setNotice] = useState("");
  const tags = useMemo(
    () => Array.from(new Set(cards.flatMap((card) => card.tags))).sort(),
    [cards],
  );
  const filteredCards = useMemo(() => {
    const query = search.trim().toLowerCase();
    return cards.filter((card) => {
      const matchesSearch =
        !query ||
        card.question.toLowerCase().includes(query) ||
        card.id.toLowerCase().includes(query) ||
        card.tags.some((cardTag) => cardTag.toLowerCase().includes(query));
      return (
        matchesSearch &&
        (type === "all" || card.type === type) &&
        (tag === "all" || card.tags.includes(tag))
      );
    });
  }, [cards, search, tag, type]);

  const saveCard = (card: LibraryCard) => {
    setCards((current) => {
      const exists = current.some((item) => item.id === card.id);
      return exists
        ? current.map((item) => (item.id === card.id ? card : item))
        : [card, ...current];
    });
    setEditorCard(null);
    setSelected(card);
    setNotice("Card saved.");
  };

  const deleteCard = (card: LibraryCard) => {
    setCards((current) => current.filter((item) => item.id !== card.id));
    setSelected(null);
    setNotice("Card deleted.");
  };

  return (
    <main className="page-shell">
      <div className="cards-page-heading">
        <PageHeading
          eyebrow="Library"
          title={t.cards}
          intro="Search, organize, preview, and back up every Card in one place."
        />
        <button className="button primary" onClick={() => setEditorCard("new")}>
          <Plus size={18} />
          Create Card
        </button>
      </div>

      {initialIntent === "upload" && (
        <div className="alert info library-notice" role="status">
          <Upload size={20} />
          <div>
            <strong>Upload your first Deck</strong>
            <p>Choose “Upload Deck” in the Files panel to add Cards to this browser.</p>
          </div>
        </div>
      )}

      {notice && (
        <div className="alert success library-notice" role="status">
          <CheckCircle2 size={20} />
          <div>
            <strong>{notice}</strong>
          </div>
          <button className="alert-close" onClick={() => setNotice("")} aria-label="Dismiss">
            <X size={18} />
          </button>
        </div>
      )}

      <div className="cards-layout">
        <section className="panel library-panel">
          <div className="library-toolbar">
            <label className="field search-field">
              <span>Search Cards</span>
              <span className="search-input">
                <Search size={18} />
                <input
                  type="search"
                  value={search}
                  onChange={(event) => setSearch(event.target.value)}
                  placeholder="Question, ID, or Tag"
                />
              </span>
            </label>
            <label className="field">
              <span>Type</span>
              <span className="select-wrap">
                <select value={type} onChange={(event) => setType(event.target.value)}>
                  <option value="all">All types</option>
                  {(
                    [
                      "qa",
                      "formula",
                      "cloze",
                      "multiple_choice",
                      "true_false",
                      "numeric",
                    ] as CardKind[]
                  ).map((kind) => (
                    <option key={kind} value={kind}>
                      {kind}
                    </option>
                  ))}
                </select>
                <ChevronDown size={18} />
              </span>
            </label>
            <label className="field">
              <span>Tag</span>
              <span className="select-wrap">
                <select value={tag} onChange={(event) => setTag(event.target.value)}>
                  <option value="all">Any tag</option>
                  {tags.map((item) => (
                    <option key={item}>{item}</option>
                  ))}
                </select>
                <ChevronDown size={18} />
              </span>
            </label>
          </div>
          <div className="library-summary">
            <span>
              {filteredCards.length} of {cards.length} Cards
            </span>
            <span>
              <SlidersHorizontal size={15} />
              Windowed list
            </span>
          </div>
          {filteredCards.length > 0 ? (
            <VirtualCardList
              key={`${search}-${type}-${tag}`}
              items={filteredCards}
              onSelect={setSelected}
            />
          ) : (
            <EmptyState compact language={language} />
          )}
          <button
            className="text-button danger-text delete-all-button"
            onClick={() => setConfirmClear(true)}
            disabled={cards.length === 0}
          >
            <Trash2 size={17} />
            {t.clear}
          </button>
        </section>

        <aside className="panel backup-panel">
          <div className="section-heading compact">
            <span className="section-icon neutral">
              <FileJson size={20} />
            </span>
            <div>
              <p className="eyebrow">Backup</p>
              <h2>Files</h2>
            </div>
          </div>
          <div className="file-grid single-column">
            <label className="button secondary file-button">
              <Upload size={18} />
              <span>{t.uploadDeck}</span>
              <input type="file" accept=".json" onChange={() => setUploadResult("success")} />
            </label>
            <label className="button secondary file-button">
              <Upload size={18} />
              <span>{t.uploadLog}</span>
              <input type="file" accept=".json" onChange={() => setUploadResult("error")} />
            </label>
            <button className="button ghost">
              <Download size={18} />
              {t.downloadCards}
            </button>
            <button className="button ghost">
              <Download size={18} />
              {t.downloadLog}
            </button>
          </div>
          <p className="backup-note">
            <CheckCircle2 size={17} />
            {t.days}
          </p>
          {uploadResult !== "none" && (
            <div
              className={uploadResult === "success" ? "alert success" : "alert danger"}
              role="status"
            >
              {uploadResult === "success" ? <CheckCircle2 size={20} /> : <AlertCircle size={20} />}
              <div>
                <strong>{uploadResult === "success" ? "Upload result" : "File rejected"}</strong>
                <p>
                  {uploadResult === "success"
                    ? "2 added, 1 updated, 0 unchanged"
                    : "card 3: en.question: Required"}
                </p>
              </div>
            </div>
          )}
        </aside>
      </div>

      {selected && (
        <Modal title="Card preview" onClose={() => setSelected(null)}>
          <div className="modal-body card-preview">
            <span className="type-label">{selected.type}</span>
            <h3>{selected.question}</h3>
            <div className="preview-answer">
              <span>Answer</span>
              <p>{selected.answer}</p>
            </div>
            <div className="row-tags">
              {selected.tags.map((item) => (
                <span key={item}>{item}</span>
              ))}
            </div>
            <p className="preview-id">
              {selected.id} · version {selected.version}
            </p>
          </div>
          <div className="modal-actions modal-actions-wrap">
            <button className="button danger" onClick={() => deleteCard(selected)}>
              <Trash2 size={17} />
              Delete
            </button>
            <button
              className="button secondary"
              onClick={() => {
                setEditorCard(selected);
                setSelected(null);
              }}
            >
              <Pencil size={17} />
              Edit
            </button>
            <button
              className="button secondary"
              onClick={() => {
                setNotice("Added to the current Session.");
                setSelected(null);
              }}
            >
              <Play size={17} />
              Add to current Session
            </button>
            <button
              className="button primary"
              onClick={() => {
                setNotice("New Session created with this Card.");
                setSelected(null);
              }}
            >
              <Plus size={17} />
              New Session
            </button>
          </div>
        </Modal>
      )}
      {editorCard && (
        <CardEditor
          card={editorCard === "new" ? undefined : editorCard}
          onSave={saveCard}
          onClose={() => setEditorCard(null)}
        />
      )}
      {confirmClear && (
        <Modal title="Delete all Cards?" onClose={() => setConfirmClear(false)}>
          <div className="modal-body">
            <div className="alert danger">
              <AlertCircle size={20} />
              <div>
                <strong>This cannot be undone.</strong>
                <p>All {cards.length} Cards will be permanently removed from this browser.</p>
              </div>
            </div>
          </div>
          <div className="modal-actions">
            <button className="button ghost" onClick={() => setConfirmClear(false)}>
              Cancel
            </button>
            <button
              className="button danger"
              onClick={() => {
                setCards([]);
                setConfirmClear(false);
              }}
            >
              Delete all Cards
            </button>
          </div>
        </Modal>
      )}
    </main>
  );
}

function StudyCard({
  card,
  state,
  onState,
  onRate,
  language,
}: {
  card: (typeof studyCards)[number];
  state: AnswerState;
  onState: (state: AnswerState) => void;
  onRate?: (state: "right" | "wrong") => void;
  language: Language;
}) {
  const t = copy[language];
  const [numeric, setNumeric] = useState("");
  const [picked, setPicked] = useState("");
  const isChoice = card.kind === "multiple_choice" || card.kind === "true_false";
  const answered = state === "right" || state === "wrong";

  const choose = (option: string) => {
    setPicked(option);
    onState(option === card.correct ? "right" : "wrong");
  };

  const checkNumeric = () => onState(numeric.trim() === card.correct ? "right" : "wrong");

  return (
    <article className={`study-card type-${card.kind}`}>
      <div className="study-card-top">
        <span className="type-label">{card.eyebrow}</span>
        <span className="card-tag">Decision Tree</span>
      </div>
      <div className="prompt">
        <h2>{card.question}</h2>
        {card.secondary && <p>{card.secondary}</p>}
      </div>

      {isChoice && (
        <div className="options" aria-label="Answer options">
          {card.options?.map((option) => {
            const status =
              answered && option === card.correct
                ? "correct"
                : answered && option === picked
                  ? "incorrect"
                  : "";
            return (
              <button
                key={option}
                className={`option-button ${status}`}
                onClick={() => !answered && choose(option)}
                disabled={answered}
              >
                <span className="option-marker">
                  {status === "correct" ? (
                    <Check size={17} />
                  ) : status === "incorrect" ? (
                    <X size={17} />
                  ) : (
                    String.fromCharCode(65 + (card.options?.indexOf(option) ?? 0))
                  )}
                </span>
                <span>
                  {card.kind === "true_false" ? (option === "True" ? t.true : t.false) : option}
                </span>
              </button>
            );
          })}
        </div>
      )}

      {card.kind === "numeric" && (
        <div className="numeric-form">
          <label className="field">
            <span>{t.answer}</span>
            <span className="input-with-unit">
              <input
                type="number"
                value={numeric}
                onChange={(event) => setNumeric(event.target.value)}
                onKeyDown={(event) => event.key === "Enter" && checkNumeric()}
                disabled={answered}
              />
              <span>ms</span>
            </span>
          </label>
          {!answered && (
            <button className="button primary" onClick={checkNumeric}>
              {t.check}
            </button>
          )}
        </div>
      )}

      {!isChoice && card.kind !== "numeric" && state === "idle" && (
        <button className="button primary large reveal-button" onClick={() => onState("revealed")}>
          {t.reveal}
        </button>
      )}

      {state === "revealed" && (
        <div className="revealed-answer">
          <span className="answer-label">{t.answer}</span>
          <p
            className={
              card.kind === "formula" ? "formula" : card.kind === "cloze" ? "cloze-answer" : ""
            }
          >
            {card.kind === "cloze" ? (
              <>
                A deeper tree has <mark>higher</mark> variance.
              </>
            ) : (
              card.answer
            )}
          </p>
          {card.kind === "formula" && <span className="example">Example: ‖(3, 4)‖ = 5</span>}
        </div>
      )}

      {answered && (
        <div className={`feedback ${state}`} role="status">
          {state === "right" ? <CheckCircle2 size={24} /> : <XCircle size={24} />}
          <div>
            <strong>{state === "right" ? t.right : t.wrong}</strong>
            <p>
              {t.answer}: {card.answer}
            </p>
            {card.explanation && <span>{card.explanation}</span>}
          </div>
        </div>
      )}

      {state === "revealed" && (
        <div className="rating-actions">
          <button
            className="button secondary"
            onClick={() => (onRate ? onRate("wrong") : onState("wrong"))}
          >
            <XCircle size={18} />
            {t.didnt}
          </button>
          <button
            className="button primary"
            onClick={() => (onRate ? onRate("right") : onState("right"))}
          >
            <CheckCircle2 size={18} />
            {t.knew}
          </button>
        </div>
      )}
    </article>
  );
}

function Session({ language, setPage }: { language: Language; setPage: (page: Page) => void }) {
  const [queue, setQueue] = useState(studyCards.map((_, index) => index));
  const [position, setPosition] = useState(0);
  const [retried, setRetried] = useState<number[]>([]);
  const [state, setState] = useState<AnswerState>("idle");
  const [complete, setComplete] = useState(false);
  const t = copy[language];
  const cardIndex = queue[position];
  const next = (outcome: AnswerState = state) => {
    let nextQueue = queue;
    if (outcome === "wrong" && !retried.includes(cardIndex)) {
      nextQueue = [...queue, cardIndex];
      setQueue(nextQueue);
      setRetried([...retried, cardIndex]);
    }
    if (position === nextQueue.length - 1) {
      setComplete(true);
    } else {
      setPosition(position + 1);
      setState("idle");
    }
  };

  return (
    <main className="session-page">
      <div className="session-toolbar">
        <button className="button ghost" onClick={() => setPage("home")}>
          <ArrowLeft size={18} />
          {t.back}
        </button>
        <div className="progress-copy">
          <span>
            {complete ? queue.length : position + 1} / {queue.length}
          </span>
          <div className="progress-track">
            <span
              style={
                {
                  "--progress": `${complete ? 100 : ((position + 1) / queue.length) * 100}%`,
                } as React.CSSProperties
              }
            />
          </div>
        </div>
      </div>
      <div className="session-stage">
        {complete ? (
          <section className="summary-card">
            <span className="summary-icon">
              <Check size={30} />
            </span>
            <p className="eyebrow">
              {language === "de" ? "Sitzung abgeschlossen" : "Session complete"}
            </p>
            <h1>{t.summary}</h1>
            <p>
              {language === "de"
                ? "Gute Arbeit. Jede Wiederholung stärkt dein Wissen."
                : "Good work. Every Review makes the next one easier."}
            </p>
            <button className="button primary large" onClick={() => setPage("home")}>
              {t.done}
              <ArrowRight size={18} />
            </button>
          </section>
        ) : (
          <>
            <StudyCard
              card={studyCards[cardIndex]}
              state={state}
              onState={setState}
              onRate={(outcome) => next(outcome)}
              language={language}
            />
            {(state === "right" || state === "wrong") && (
              <button className="button primary large next-button" onClick={() => next()}>
                {t.next}
                <ArrowRight size={18} />
              </button>
            )}
          </>
        )}
      </div>
    </main>
  );
}

    </div>
  );
}
